use sst_tgi

// RQF-005: Modificar rol de Usuario

db.usuario.updateOne(
  { nombreUsuario: "Don Pepe"},
  {$set: {rolUsuario: "ayudante"}}
)

// RQF-006: Modificar estado de Usuario

db.usuario.updateOne(
  { nombreUsuario: "Don Pepe"},
  {$set: {estadoUsuario: "inactivo"}}
)

// Para retificar que los anteriores 2 RQFs corrieron bien:
db.usuario.findOne({nombreUsuario: "Don Pepe"})

// RQF-045: Registrar asistencia a capacitación

db.asistencia.updateOne(
  { idProgramacionFK: 1, idTrabajadorFK: 1, asistio: false },
  {$set: {asistio: true, 
          evaluacionCapacitador: 5, 
          comentariosCapacitacion: "Estuvo ok"},
   $currentDate: { fechaDeAsistencia: true }
  }
)

db.asistencia.findOne({idProgramacionFK: 1, idTrabajadorFK: 1})

// Nota para el futuro: El segundo {} del update es para pasarle muchas instrucciones, $set es apenas una de ellas.
